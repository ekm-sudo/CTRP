%pyspark
last_top_up = F.max(F.when(F.col('topped_up') == 1, F.col('adate')))
days = hist.groupBy('taxpayer_tin').agg(F.datediff(snap_date, last_top_up).alias('par105_days_since_last_top_up'))
parts['par105_top_up_180d'] = days.withColumn(
    'par105_top_up_180d', F.when(F.col('par105_days_since_last_top_up') <= 180, 1.0).otherwise(0.0))
peek(parts['par105_top_up_180d'])
